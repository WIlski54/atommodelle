// PDF Export function exportToPDF() { try { // Prüfe ob jsPDF geladen ist if (typeof window.jspdf === 'undefined') { alert('PDF-Bibliothek wird geladen... Bitte warten Sie einen Moment und versuchen Sie es erneut.'); return; } const { jsPDF } = window.jspdf; const doc = new jsPDF({ orientation: 'p', unit: 'mm', format: 'a4' }); let yPos = 20; const leftMargin = 20; const contentWidth = doc.internal.pageSize.getWidth() - 2 * leftMargin; function addHeader() { doc.setFontSize(18); doc.setTextColor(102, 126, 234); doc.text('Ergebnisse: Atombau und Periodensystem', doc.internal.pageSize.getWidth() / 2, yPos, { align: 'center' }); yPos += 15; } function checkYPos(requiredSpace = 20) { if (yPos > (doc.internal.pageSize.getHeight() - requiredSpace)) { doc.addPage(); yPos = 20; } } addHeader(); doc.setFontSize(12); doc.setTextColor(0, 0, 0); doc.text(`Erreichte Punkte: ${points} / ${maxPoints}`, leftMargin, yPos); yPos += 8; doc.text(`Datum: ${new Date().toLocaleDateString('de-DE')}`, leftMargin, yPos); yPos += 15; doc.setDrawColor(200); doc.line(leftMargin, yPos, doc.internal.pageSize.getWidth() - leftMargin, yPos); yPos += 10; function addSection(title, content) { checkYPos(20); doc.setFontSize(14); doc.setTextColor(118, 75, 162); doc.text(title, leftMargin, yPos); yPos += 8; doc.setFontSize(10); doc.setTextColor(0, 0, 0); const lines = doc.splitTextToSize(content, contentWidth); lines.forEach(line => { checkYPos(10); doc.text(line, leftMargin, yPos); yPos += 6; }); yPos += 5; } // Aufgabe 1: Grundwissen addSection('Aufgabe 1: Grundwissen', 'Deine Antworten wurden registriert.'); yPos += 3; // Aufgabe 2: Rechenaufgaben doc.setFontSize(14); doc.setTextColor(118, 75, 162); checkYPos(30); doc.text('Aufgabe 2: Rechnen mit Atommassen', leftMargin, yPos); yPos += 8; doc.setFontSize(10); doc.setTextColor(0, 0, 0); const calc1 = document.getElementById('calc1').value || '(nicht ausgefüllt)'; const calc2 = document.getElementById('calc2').value || '(nicht ausgefüllt)'; const calc3 = document.getElementById('calc3').value || '(nicht ausgefüllt)'; doc.text(`a) Kohlenstoff: ${calc1} u`, leftMargin, yPos); yPos += 6; doc.text(`b) Sauerstoff: ${calc2} Neutronen`, leftMargin, yPos); yPos += 6; doc.text(`c) Natrium: ${calc3} u`, leftMargin, yPos); yPos += 10; // Aufgabe 3: Elektronenverteilung doc.setFontSize(14); doc.setTextColor(118, 75, 162); checkYPos(40); doc.text('Aufgabe 3: Elektronenverteilung', leftMargin, yPos); yPos += 8; doc.setFontSize(10); doc.setTextColor(0, 0, 0); const shell1a = document.getElementById('shell1a').value || '?'; const shell2a = document.getElementById('shell2a').value || '?'; doc.text(`a) Stickstoff: 1. Schale: ${shell1a}, 2. Schale: ${shell2a}`, leftMargin, yPos); yPos += 6; const shell1b = document.getElementById('shell1b').value || '?'; const shell2b = document.getElementById('shell2b').value || '?'; const shell3b = document.getElementById('shell3b').value || '?'; doc.text(`b) Magnesium: 1. Schale: ${shell1b}, 2. Schale: ${shell2b}, 3. Schale: ${shell3b}`, leftMargin, yPos); yPos += 6; const shell1c = document.getElementById('shell1c').value || '?'; const shell2c = document.getElementById('shell2c').value || '?'; const shell3c = document.getElementById('shell3c').value || '?'; doc.text(`c) Argon: 1. Schale: ${shell1c}, 2. Schale: ${shell2c}, 3. Schale: ${shell3c}`, leftMargin, yPos); yPos += 10; // Aufgabe 4: Zuordnungen doc.setFontSize(14); doc.setTextColor(118, 75, 162); checkYPos(40); doc.text('Aufgabe 4: Begriffe zuordnen', leftMargin, yPos); yPos += 8; doc.setFontSize(10); doc.setTextColor(0, 0, 0); const zones = document.querySelectorAll('.drop-zone'); const zoneNames = ['Protonen', 'Neutronen', 'Elektronen']; zones.forEach((zone, index) => { const terms = Array.from(zone.querySelectorAll('.placed-term')).map(t => t.textContent).join(', '); const lines = doc.splitTextToSize(`${zoneNames[index]}: ${terms || '(keine Zuordnung)'}`, contentWidth - 5); lines.forEach(line => { checkYPos(10); doc.text(line, leftMargin, yPos); yPos += 6; }); }); yPos += 5; // Aufgabe 5: Sätze doc.setFontSize(14); doc.setTextColor(118, 75, 162); checkYPos(20); doc.text('Aufgabe 5: Ausformulierte Sätze', leftMargin, yPos); yPos += 8; const particleTypes = ['Protonen', 'Neutronen', 'Elektronen']; let sentenceIndex = 1; particleTypes.forEach(type => { checkYPos(25); doc.setFont(undefined, 'bold'); doc.setFontSize(11); doc.text(`Sätze über ${type}:`, leftMargin, yPos); yPos += 7; doc.setFont(undefined, 'normal'); doc.setFontSize(10); for (let i = 0; i < 3; i++) { const text = document.getElementById(`s${sentenceIndex}`).value || '(nicht ausgefüllt)'; const lines = doc.splitTextToSize(`${i + 1}. ${text}`, contentWidth - 5); lines.forEach(line => { checkYPos(10); doc.text(line, leftMargin + 5, yPos); yPos += 6; }); sentenceIndex++; } yPos += 4; }); // Footer const pageCount = doc.internal.getNumberOfPages(); for(let i = 1; i <= pageCount; i++) { doc.setPage(i); doc.setFontSize(8); doc.setTextColor(128, 128, 128); doc.text(`Seite ${i} von ${pageCount}`, doc.internal.pageSize.getWidth() / 2, 285, { align: 'center' }); doc.text('Erstellt mit dem interaktiven Arbeitsblatt Atombau', leftMargin, 285); } // Speichern mit iOS-Kompatibilität try { doc.save('Atombau_Ergebnisse.pdf'); } catch (saveError) { // Alternative: PDF als Blob öffnen (funktioniert besser auf iOS) const pdfBlob = doc.output('blob'); const url = URL.createObjectURL(pdfBlob); const link = document.createElement('a'); link.href = url; link.download = 'Atombau_Ergebnisse.pdf'; link.click(); setTimeout(() => URL.revokeObjectURL(url), 100); } alert('PDF wurde erfolgreich erstellt!'); } catch (error) { console.error('Fehler beim PDF-Export:', error); alert('Fehler beim PDF-Export: ' + error.message + '\n\nBitte versuchen Sie es erneut oder kontaktieren Sie Ihren Lehrer.'); } }
